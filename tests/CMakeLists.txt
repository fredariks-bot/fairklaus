#--- GUI-Smoke-Test (nur Linux mit Editor-Build; CI fuehrt ihn via xvfb-run aus)
if(FAIRKLAUS_EDITOR_AVAILABLE AND NOT WIN32 AND X11_FOUND)
    # direkt gegen die Plugin-Quellen linken (eine vstgui-Kopie, RunLoop::init normal moeglich)
    add_executable(gui_smoke gui_smoke.cpp ${CMAKE_SOURCE_DIR}/src/PunishChild670.cpp ${CMAKE_SOURCE_DIR}/src/editor/Editor.cpp)
    target_include_directories(gui_smoke PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/vst3sdk ${CMAKE_SOURCE_DIR}/vst3sdk/vstgui4)
    target_compile_definitions(gui_smoke PRIVATE FAIRKLAUS_WITH_EDITOR=1)
    target_compile_features(gui_smoke PRIVATE cxx_std_17)
    target_link_libraries(gui_smoke PRIVATE sdk vstgui vstgui_support ${X11_LIBRARIES})
    find_program(XVFB_RUN_EXEC xvfb-run)
    if(XVFB_RUN_EXEC)
        add_test(NAME gui_smoke COMMAND ${XVFB_RUN_EXEC} -a $<TARGET_FILE:gui_smoke>)
    endif()
endif()
